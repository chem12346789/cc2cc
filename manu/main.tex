\section{Introduction}
The well-known Kohn--Sham equation \cite{kohnSelfConsistentEquationsIncluding1965-11-15} is
\begin{align}
	\label{eq:KS}
	\Big( - \frac{1}{2} \nabla^{2} + v_{\mathrm{ext}} + v_{\mathrm{H}} + v_{\rm{xc}}\Big) \phi_{i} = \epsilon_{i} \phi_{i}.
\end{align}
% 
Where $\phi_{i}$ is the Kohn--Sham orbital, $\epsilon_{i}$ is the Kohn--Sham orbital energy, $v_{\mathrm{ext}}$ is the external potential (such as the electrostatic potential of nuclei), $v_{\mathrm{H}}$ is the electrostatic potential of the electron, and $v_{\rm{xc}}$ is the exchange-correlation potential.
% 
According to the Hohenberg-Kohn theorem \cite{hohenbergInhomogeneousElectronGas1964-11-09}, exchange-correlation energy density $e_{\rm{xc}}$ is the unique functional of the ground-state electron density, $\rho = n_{i} \vert \phi_{i} \vert^{2}$, where $n_{i}$ is the occupation number of the Kohn-Sham orbital.
%
Then, the Kohn-Sham equation is solved iteratively utilizing the so-called potential term $v_{\rm{xc}} = \delta E_{\rm{xc}} / \delta \rho$.
% 
So, once this unique functional is obtained, the properties of the molecules can be calculated accurately and efficiently.
%
This is the main idea of the density functional theory (DFT) \cite{parrDensityFunctionalTheory1980}.

One of the challenging tasks of the DFT calculation is to find the unique functional between the exact exchange-correlation energy density $e_{\rm{xc}}$ and the electron density $\rho$.
% 
Numerous approximations have been proposed for this functional, including early local-density approximation functionals \cite{parrDensityFunctionalTheory1980}, hybrid functionals \cite{perdewAtomsMoleculesSolids1992-09-15}, and the rapidly growing class of machine-learning (ML) functionals \cite{nagaiCompletingDensityFunctional2020-05-05,chenDeePKSComprehensiveDataDriven2021-01-12,kirkpatrickPushingFrontiersDensity2021-12-10,luiseAccurateScalableExchangecorrelation2025-06-23,anMitigatingErrorCancellation2025-08-07}.
% 



\section{The NEURAL NETWORK}
We use the transformer \cite{vaswaniAttentionAllYou2023-08-02} as the neural network to learn the functional between the electron density and the energy density of the molecule.
% 
After the training, the neural network can predict the energy of the molecule given the electron density and then can be used to optimize the electron density iteratively using the self-consistent field (SCF) method.

The architecture of the neural network is shown in \Fig{fig:arch}.
% 
The total number of parameters in the neural network is around $1.2$ million.
%
The input of the neural network is a set of the energy density ${\tilde{\bm{e}}_{\rm{cube}}(\brm{r}_i)}$ with the size of $(N, 27, 4)$, and the output is the energy density ${e(\brm{r}_i)}$ with the size of $(N, 1)$.
% 
N is the number of the grid points in the normal dft procedure.
%
\begin{align}
	\tilde{\bm{e}}_{\rm{cube}}(\brm{r}_{i}) =
	 & \big\{ \tilde{e}(\mathbf{r}'_{i}) \text{ for } \mathbf{r}'_{i} \in \mathbf{r}_{i;\rm{cube}} \big\}.
\end{align}
The $\tilde{e}(\brm{r})$ is the energy density calculated by the 4 functionals (LDA, VWN3, B88 and LYP) from the original B3LYP functional \cite{perdewAtomsMoleculesSolids1992-09-15}.
% 
The $\mathbf{r}_{i;\rm{cube}}$ is a cube around the grid point $\mathbf{r}_{i}$ with the size of $3 \times 3 \times 3$ and the distance between two adjacent grid points is $d$.
%
This means that for each grid point $\mathbf{r}_{i}$, we consider a small region around it to capture the local environment and provide more features for the neural network.
% 
We refer to the grid points within the cube, excluding the central point, as auxiliary grid points. Details of their contributions to the energy density are provided in the Appendix.
% 
The energy density at the grid point $\mathbf{r}_{i}$, denoted as $e(\brm{r}_{i})$, can be predicted by the neural network as
% 
\begin{align}
	e(\brm{r}_{i})
	= & {\rm{NN}}\big( \tilde{\bm{e}}_{\rm{cube}}(\brm{r}_{i}); \w \big) \tilde{\bm{e}}_{\rm{B3LYP}}(\brm{r}_{i}) \nl
	= & {\rm{NN}}\big( \tilde{\bm{e}}_{\rm{cube}}(\brm{r}_{i}); \w \big)  \big( 0.08 e_{\rm{LDA}}(\brm{r}_{i}) \nl
	  & + 0.19 e_{\rm{VWN3}}(\brm{r}_{i}) + 0.72 e_{\rm{B88}}(\brm{r}_{i}) \nl
	  & + 0.81 e_{\rm{LYP}}(\brm{r}_{i}) \big),
\end{align}
% 
where ${\rm{NN}}(\tilde{\bm{e}}_{\rm{cube}}(\brm{r}_{i}); \w)$ is the output of neural network with the parameters $\w$ and input $\tilde{\bm{e}}_{\rm{cube}}(\brm{r}_{i})$, $\tilde{\bm{e}}_{\rm{B3LYP}}(\brm{r}_{i})$ is the energy density calculated by the original B3LYP functional \cite{perdewAtomsMoleculesSolids1992-09-15}.
% 
The total exchange-correlation energy is obtained by integrating the energy density over the entire grid.
% 
Then the exchange-correlation potential can be calculated by the functional derivative of the exchange-correlation energy with respect to the electron density, which is used in the Kohn-Sham equation \eq{eq:KS}.

\subsection{Dataset and the loss function}
The dataset we use to train the neural network is listed in the Appendix.
%
We use 128 molecules which consistent of less than 3 heavy atoms as the training set and 115 molecules contain 4-5 heavy atoms as the validation set.
% 
In this article, we use the CCSD(T)/Def2-QZVP (except the anions) and CCSD(T)/Ma-Def2-QZVP (the anions) \cite{bartlettCoupledclusterTheoryQuantum2007-02-22,weigendBalancedBasisSets2005,zhengMinimallyAugmentedKarlsruhe2011-02} to calculate the electron density and the energy density of the molecules.
%
The electron density will be used as the calculation of the input of the neural network and the energy density will be used as the target of the neural network.

The loss function is defined as
\begin{align}
	\label{eq:loss}
	L_{e} =
	 & \lambda_{\rm{RE}} N_{\rm{atom}} \Big(\int \d \brm{r}_{i} \big(\bar{e}(\brm{r}_{i}) - e(\brm{r}_{i})\big) \Big)^2 \nl
	 & + \lambda_{\rm{ABS}} N_{\rm{atom}} \Big(\int \d \brm{r}_{i} \Big| \bar{e}(\brm{r}_{i}) - e(\brm{r}_{i}) \Big| \Big)^2 \nl
	 & + \lambda_{\rm{AE}} N_{\rm{atom}} \Big(\bar{E}^{\rm{AE}}(\brm{r}_{i}) - E^{\rm{AE}}(\brm{r}_{i}) \Big)^2 .
\end{align}
\begin{align}
	\label{eq:loss_self_scaled}
	L_{e} =
	 & N_{\rm{atom}} \frac{\Big(\int \d \brm{r}_{i} \big(\bar{e}(\brm{r}_{i}) - e(\brm{r}_{i})\big) \Big)^2}{\Big( \int \d \brm{r}_{i} \big| \bar{e}(\brm{r}_{i}) \big| \Big)^{2}} \nl
	 & + \lambda_{\rm{ABS}} N_{\rm{atom}} \frac{\Big(\int \d \brm{r}_{i} \Big| \bar{e}(\brm{r}_{i}) - e(\brm{r}_{i}) \Big| \Big)^2}{\Big( \int \d \brm{r}_{i} \big| \bar{e}(\brm{r}_{i}) \big| \Big)^{2}} \nl
	 & + N_{\rm{atom}} \frac{\Big(\bar{E}^{\rm{AE}}(\brm{r}_{i}) - E^{\rm{AE}}(\brm{r}_{i}) \Big)^2}{\Big( \int \d \brm{r}_{i} \big| \bar{E}^{\rm{AE}}(\brm{r}_{i}) \big| \Big)^{2}} .
\end{align}
where $N_{\rm{atom}}$ is the number of atoms in the molecule and $E^{\rm{AE}}$ and $\bar{E}^{\rm{AE}}$ is the atomic energy of the predicted and target values, respectively.
%
The $\lambda_{\rm{RE}}$, $\lambda_{\rm{ABS}}$, and $\lambda_{\rm{AE}}$ are the weights of the three terms in the loss function.
%
The first term is the relative error in the total energy, the second term is the absolute error in the total energy, and the third term is the error in the atomic energy.

\subsection{Training details}
The neural network is trained using the AdamW optimizer \cite{loshchilovDecoupledWeightDecay2019-01-04} with the scheduler of cosine annealing with warm restarts \cite{loshchilovSGDRStochasticGradient2017-05-03}.
%
We set the initial learning rate to be $1 \times 10^{-4}$ and then gradually decrease it to $0$ using the cosine annealing scheduler, with a restart every $160$ epochs and a multiplication factor of $2$ for the initial learning rate after each restart.
%
The parameters of the AdamW optimizer are set to be $\beta_1 = 0.9$, $\beta_2 = 0.999$, and $\epsilon = 1 \times 10^{-8}$, and the weight decay is set to be $1 \times 10^{-12}$.
% 
The weights in the loss function \eq{eq:loss} are set to be $\lambda_{\rm{ABS}} = 0.01$.
%
We use a batch size of $4$ and train the model for $10000$ epochs.
%
The training process is monitored using the validation set every $50$ epochs, and the model with the lowest validation loss is selected as the final model.

The training is performed on a single NVIDIA A100 GPU with 80GB memory with double precision training using PyTorch \cite{paszkePyTorchImperativeStyle2019-12-03} and the training takes around $96$ hours to complete. The version of PyTorch used is $2.8.0$, and CUDA version is $12.2$, the version of driver is $535.129.03$.

We use the energy density calculated by the four functionals (LDA, VWN3, B88, and LYP) from the original B3LYP functional \cite{perdewAtomsMoleculesSolids1992-09-15} as the input features of the neural network, not the electron density and the derivatives of the electron density directly.
% 
\begin{align}
	\tilde{e}(\brm{r}) =
	 & \big( e_{\rm{LDA}}(\brm{r}), e_{\rm{VWN3}}(\brm{r}), e_{\rm{B88}}(\brm{r}), e_{\rm{LYP}}(\brm{r}) \big).
\end{align}
%
This choice is motivated by the very wide dynamic range of the electron density, $\rho$, and its derivatives, which complicates learning and can destabilize both the optimization and the validation process.
%
By using the energy densities from established components (LDA, VWN3, B88, LYP) as input features, we provide numerically stable, physically informed representations of the local electronic environment.

The network outputs the residual energy density, defined as $e(\brm{r}) = e_{\rm{xc}}^{\rm{CCSD(T)}}(\brm{r}) - e_{\rm{xc}}^{\rm{B3LYP}}(\brm{r})$, where the B3LYP term is computed from the input features and the CCSD(T) term serves as the target.
% 
The total exchange-correlation energy density is then reconstructed by adding the B3LYP energy density back to the predicted residual, $E_{\rm{xc}}^{\rm{pred}} = E'_{\rm{xc}} + E_{\rm{xc}}^{\rm{B3LYP}}$.

\section{Auxiliary Grid Points}
To provide additional features for the neural network, we consider a small cube surrounding each grid point $\mathbf{r}_{i} = (x_{i},y_{i},z_{i})$.
% 
This cube has dimensions of $3 \times 3 \times 3$, with $d$ representing the distance between two adjacent grid points.
% 
To make it clear, the auxiliary grid points $\mathbf{r}_{i;\rm{aux}}$ are defined as all grid points within this cube, excluding the central point $\mathbf{r}_{i}$ itself.
\begin{align}
	\mathbf{r}_{i;\rm{aux}} =
	 & \big\{ \mathbf{r}'_{i} \in \mathbf{r}_{i;\rm{cube}} \text{ and } \mathbf{r}'_{i} \neq \mathbf{r}_{i} \big\}.
\end{align}

In the standard DFT procedure, each grid point is assigned a weight $w_{i}$, which is used to perform numerical integration.
% 
We set all the weights of the auxiliary grid points ($w_{i;\rm{aux}}$) to be zero. Thus, they do not contribute to the total energy and potential in the DFT procedure.
% 
\begin{align}
	E'_{\rm{xc}} =
	  & \int e(\mathbf{r}_{i;\rm{cube}}) \d \mathbf{r}_{i;\rm{cube}} \nl
	= & \sum_{i} w_{i} e(\brm{r}_{i}) + \sum_{i} \sum_{\brm{r}'_{i} \in \brm{r}_{i;\rm{aux}}} w_{i;\rm{aux}} e(\brm{r}'_{i}) \nl
	= & \sum_{i} w_{i} e(\brm{r}_{i}).
\end{align}
% 
As a result, the auxiliary grid points influence the potential but do not affect the fock matrix or the total energy. It makes the procedure remains consistent with the standard DFT calculation.
% 
\begin{align}
	v'_{\rm{xc}}(\brm{r}_{i}) =
	  & \frac{\delta E'_{\rm{xc}}}{\delta \rho(\brm{r}_{i})} \nl
	= & \sum_{j} w_{j} \frac{\delta e(\brm{r}_{j})}{\delta \rho(\brm{r}_{i})} \nl
	= & \sum_{j} w_{j} \frac{\delta {\rm{NN}}\big( \tilde{\bm{e}}_{\rm{cube}}(\brm{r}_{i}); \w \big)}{\delta \rho(\brm{r}_{i})} \nl
	= & \sum_{j} w_{j} \frac{\partial {\rm{NN}}\big( \tilde{\bm{e}}_{\rm{cube}}(\brm{r}_{i}); \w \big)}{\partial \tilde{\bm{e}}_{\rm{cube}}(\brm{r}_{i})} \frac{\delta \tilde{\bm{e}}_{\rm{cube}}(\brm{r}_{i})}{\delta \rho(\brm{r}_{i})}
\end{align}

To make the neural network can capture the rotational variance, we set the cube to be aligned with the main rotation axis of the secondary derivative matrix of the electron density at the grid point $\brm{r}_{i}$.
%
This secondary derivative matrix is defined as
\begin{align}
	H(\brm{r}_{i}) =
	 & \begin{pmatrix}
		   \frac{\partial^{2} \rho(\brm{r}_{i})}{\partial x^{2}}        & \frac{\partial^{2} \rho(\brm{r}_{i})}{\partial x \partial y} & \frac{\partial^{2} \rho(\brm{r}_{i})}{\partial x \partial z} \\
		   \frac{\partial^{2} \rho(\brm{r}_{i})}{\partial y \partial x} & \frac{\partial^{2} \rho(\brm{r}_{i})}{\partial y^{2}}        & \frac{\partial^{2} \rho(\brm{r}_{i})}{\partial y \partial z} \\
		   \frac{\partial^{2} \rho(\brm{r}_{i})}{\partial z \partial x} & \frac{\partial^{2} \rho(\brm{r}_{i})}{\partial z \partial y} & \frac{\partial^{2} \rho(\brm{r}_{i})}{\partial z^{2}}
	   \end{pmatrix}.
\end{align}
% 
The eigenvectors $\mathbf{u}(\brm{r}_{i})$ of this matrix represent the main rotation axes, and we align the cube with these axes.
% 
\begin{align}
	\mathbf{r}_{i;\rm{cube}} =
	 & \big\{ \mathbf{r}_{i} + d (m \mathbf{u}_{1}(\brm{r}_{i}) + n \mathbf{u}_{2}(\brm{r}_{i}) + p \mathbf{u}_{3}(\brm{r}_{i})) \nl
	 & \text{ for } m,n,p \in \{-1,0,1\} \big\}.
\end{align}
